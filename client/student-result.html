const resultForm = document.getElementById("resultForm"); const studentIdInput = document.getElementById("studentId"); const resultMessage = document.getElementById("resultMessage"); if (resultForm) { resultForm.addEventListener("submit", async function (event) { event.preventDefault(); const studentId = studentIdInput.value.trim(); if (!studentId) { resultMessage.innerHTML = "

Please enter your Student ID.

"; return; } resultMessage.innerHTML = "

Loading result...

"; try { const response = await fetch(`/api/results/${studentId}`); const data = await response.json(); if (!response.ok) { resultMessage.innerHTML = `

${data.error || "Result not found."}

`; return; } let total = 0; data.results.forEach(result => { total += Number(result.score); }); const average = total / data.results.length; resultMessage.innerHTML = `

Student Result

Student ID: ${data.studentId}

${data.results.map(result => { const score = Number(result.score); let grade; if (score >= 70) { grade = "A"; } else if (score >= 60) { grade = "B"; } else if (score >= 50) { grade = "C"; } else if (score >= 45) { grade = "D"; } else if (score >= 40) { grade = "E"; } else { grade = "F"; } return ` `; }).join("")}
Subject Score Grade
${result.subject} ${score} ${grade}

Total: ${total}

Average: ${average.toFixed(2)}

`; } catch (error) { console.error(error); resultMessage.innerHTML = `

Unable to connect to the server.

`; } }); }
